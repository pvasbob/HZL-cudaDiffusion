cmake_minimum_required(VERSION 3.22)
project(HZLDiffusion VERSION 0.1.0 LANGUAGES CXX)
option(HZL_CUDA "Build explicit CUDA backend" OFF)
option(HZL_TESTS "Build validation tests" ON)
option(HZL_BENCHMARKS "Build forecast benchmark" ON)
set(HZL_CXX_STANDARD 20 CACHE STRING "Host C++ standard (20 or 23)")
if(NOT HZL_CXX_STANDARD MATCHES "^(20|23)$")
  message(FATAL_ERROR "HZL_CXX_STANDARD must be 20 or 23")
endif()
set(CMAKE_CXX_STANDARD ${HZL_CXX_STANDARD})
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(HZL_SANITIZERS "CPU AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
if(HZL_SANITIZERS)
  if(HZL_CUDA)
    message(FATAL_ERROR "Use a separate CPU build for host sanitizers")
  endif()
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_library(hzl src/model.cpp src/observation.cpp src/control.cpp src/experiment.cpp)
target_include_directories(hzl PUBLIC include)
target_compile_options(hzl PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Wpedantic>)
if(HZL_CUDA)
  set(CMAKE_CUDA_ARCHITECTURES 86 CACHE STRING "CUDA architectures")
  enable_language(CUDA)
  find_package(CUDAToolkit 12.0 REQUIRED)
  # CMake 3.22 predates NVCC C++20 flag discovery; NVCC 12 supports it.
  if(CMAKE_VERSION VERSION_LESS 3.25)
    set(CMAKE_CUDA20_STANDARD_COMPILE_OPTION "-std=c++20")
    set(CMAKE_CUDA20_EXTENSION_COMPILE_OPTION "-std=c++20")
  endif()
  set_property(TARGET hzl PROPERTY CUDA_STANDARD 20)
  set_property(TARGET hzl PROPERTY CUDA_STANDARD_REQUIRED ON)
  set_property(TARGET hzl PROPERTY CUDA_ARCHITECTURES "${CMAKE_CUDA_ARCHITECTURES}")
  target_sources(hzl PRIVATE src/solver.cu src/cuda_image.cu src/cuda_forecast.cu)
  target_compile_definitions(hzl PUBLIC HZL_HAS_CUDA=1)
  target_link_libraries(hzl PUBLIC CUDA::cudart)
endif()
option(HZL_OPENCV_IMAGE "Use OpenCV CPU Gaussian filtering" ON)
if(HZL_OPENCV_IMAGE)
  find_package(OpenCV 4.5 REQUIRED COMPONENTS core imgproc)
  target_include_directories(hzl PRIVATE ${OpenCV_INCLUDE_DIRS})
  target_link_libraries(hzl PRIVATE ${OpenCV_LIBS})
  target_compile_definitions(hzl PRIVATE HZL_OPENCV_IMAGE=1)
endif()
option(HZL_OPENCV_OPTIM "Use OpenCV Nelder-Mead optimizer" ON)
option(HZL_GUI "Build native OpenCV visualization" OFF)
if(HZL_OPENCV_OPTIM)
  find_package(OpenCV 4.5 REQUIRED COMPONENTS core)
  target_include_directories(hzl PRIVATE ${OpenCV_INCLUDE_DIRS})
  target_link_libraries(hzl PRIVATE ${OpenCV_LIBS})
  target_compile_definitions(hzl PUBLIC HZL_OPENCV_OPTIM=1)
endif()
if(HZL_GUI)
  find_package(OpenCV 4.5 REQUIRED COMPONENTS core imgproc imgcodecs highgui)
  add_executable(hzl-gui src/gui.cpp)
  target_include_directories(hzl-gui PRIVATE ${OpenCV_INCLUDE_DIRS})
  target_link_libraries(hzl-gui PRIVATE hzl ${OpenCV_LIBS})
endif()
add_executable(hzl-control src/control_main.cpp)
target_link_libraries(hzl-control PRIVATE hzl)
add_executable(hzl-diffusion src/main.cpp)
target_link_libraries(hzl-diffusion PRIVATE hzl)
if(HZL_BENCHMARKS)
  find_package(OpenCV 4.5 QUIET COMPONENTS core imgproc)
  if(OpenCV_FOUND)
    add_executable(hzl-compare src/compare.cpp)
    target_include_directories(hzl-compare PRIVATE ${OpenCV_INCLUDE_DIRS})
    target_link_libraries(hzl-compare PRIVATE hzl ${OpenCV_LIBS})
  endif()
  add_executable(hzl-benchmark src/benchmark.cpp)
  target_link_libraries(hzl-benchmark PRIVATE hzl)
endif()
add_executable(hzl-observe src/observe_main.cpp)
target_link_libraries(hzl-observe PRIVATE hzl)
option(HZL_OPENCV_REFERENCE "Validate reconstruction against installed OpenCV" OFF)
if(HZL_TESTS)
  enable_testing()
  add_executable(hzl-tests tests/numerics.cpp)
  target_link_libraries(hzl-tests PRIVATE hzl)
  add_test(NAME numerical-validation COMMAND hzl-tests)
  add_executable(hzl-control-tests tests/control.cpp)
  target_link_libraries(hzl-control-tests PRIVATE hzl)
  add_test(NAME control-validation COMMAND hzl-control-tests)
  add_test(NAME configured-run COMMAND hzl-control --config ${CMAKE_CURRENT_SOURCE_DIR}/examples/default.cfg --endpoint 1 --interval 0.5 --budget 8 --method finite --output ${CMAKE_CURRENT_BINARY_DIR}/configured-test)
  add_test(NAME invalid-config COMMAND hzl-control --interval 0)
  set_tests_properties(invalid-config PROPERTIES WILL_FAIL TRUE)
  if(HZL_GUI)
    add_test(NAME dashboard-interaction COMMAND hzl-gui --interaction-smoke --width 17 --height 13 --endpoint 2 --interval 1 --budget 8 --method finite --output ${CMAKE_CURRENT_BINARY_DIR}/interaction-test)
    add_test(NAME dashboard-render COMMAND hzl-gui --render-only --width 17 --height 13 --endpoint 1 --budget 8 --output ${CMAKE_CURRENT_BINARY_DIR}/dashboard-test)
  endif()
  add_executable(hzl-observation-tests tests/observations.cpp)
  target_link_libraries(hzl-observation-tests PRIVATE hzl)
  if(HZL_OPENCV_REFERENCE)
    find_package(OpenCV 4.5 REQUIRED COMPONENTS core imgproc)
    target_include_directories(hzl-observation-tests PRIVATE ${OpenCV_INCLUDE_DIRS})
    target_link_libraries(hzl-observation-tests PRIVATE ${OpenCV_LIBS})
    target_compile_definitions(hzl-observation-tests PRIVATE HZL_OPENCV_REFERENCE=1)
  endif()
  add_test(NAME observation-validation COMMAND hzl-observation-tests)
  if(HZL_CUDA)
    add_executable(hzl-cuda-variant-tests tests/cuda_variants.cpp)
    target_link_libraries(hzl-cuda-variant-tests PRIVATE hzl)
    add_test(NAME cuda-variants COMMAND hzl-cuda-variant-tests)
    set_tests_properties(cuda-variants PROPERTIES SKIP_RETURN_CODE 77)
    add_test(NAME cuda-resident-control-validation COMMAND hzl-control-tests --cuda --resident)
    set_tests_properties(cuda-resident-control-validation PROPERTIES SKIP_RETURN_CODE 77)
    add_test(NAME cuda-control-validation COMMAND hzl-control-tests --cuda)
    set_tests_properties(cuda-control-validation PROPERTIES SKIP_RETURN_CODE 77)
    add_test(NAME cuda-observation-validation COMMAND hzl-observation-tests --cuda)
    set_tests_properties(cuda-observation-validation PROPERTIES SKIP_RETURN_CODE 77)
    add_test(NAME cuda-validation COMMAND hzl-tests --cuda)
    set_tests_properties(cuda-validation PROPERTIES SKIP_RETURN_CODE 77)
  endif()
endif()
